<!-- WIDGETDESCRIPTION:
A shared color swatch that stays in sync across every board instance using the Alleo synced-status API.
-->
<!-- WIDGETHELP:
## What it does

This widget stores a single shared color in the Alleo synced status, so every
person viewing the board sees the same swatch at the same time. Tap **Randomize**
to pick a new color; the change is written to synced status and instantly
mirrored to all other instances.

## How to use it

1. Enable **Enable Alleo board features** and **Enable synchronizing status** in
   the widget settings.
2. Tap **Randomize** to choose a new shared color.
3. Open the same board in another window to watch both swatches update together.

You can also drive the widget externally: send `set-color` with a `color`
parameter (any CSS color) to set a specific color, or `reset` to return to the
default.

## Settings

Board editors can adjust the background and text colors from the widget settings.
The synced-status key and default color live in the configuration section at the
top of the file.

## Notes

The synced status is the single source of truth: the swatch always reflects the
last value written by any instance, and newly opened instances render the shared
color immediately on load.
-->

        <script>
            /**
             * Stores a single shared color in synced status so every board viewer
             * sees the same value. The synced status is the source of truth.
             */
            class SyncedColorWidget {
                /* =============== Configuration =============== */

                /** @type {string} Key under which the shared color is stored. */
                statusKey = settings.statusKey

                /** @type {string} Fallback color used before status arrives or after reset. */
                defaultColor = settings.defaultColor

                /** @type {string} Label shown on the randomize button. */
                buttonLabel = settings.buttonLabel

                /* =============== End Configuration =============== */

                /**
                 * Stores DOM references used by the controller.
                 * @param {HTMLElement} swatch Swatch element whose background reflects the color.
                 * @param {HTMLElement} valueElement Element showing the color value.
                 * @param {HTMLButtonElement} button Randomize button.
                 */
                constructor(swatch, valueElement, button) {
                    /** @type {HTMLElement} Swatch element. */
                    this.swatch = swatch
                    /** @type {HTMLElement} Element showing the color value. */
                    this.valueElement = valueElement
                    /** @type {HTMLButtonElement} Randomize button. */
                    this.button = button
                }

                /**
                 * Subscribes to shared state, wires events, and initializes the SDK
                 * so the initial synced color is applied before onloaded fires.
                 */
                async start() {
                    this.button.textContent = this.buttonLabel
                    alleo.onSyncedStatusUpdate((status) => this.applyStatus(status))
                    alleo.onIncomingAction((payload) => this.handleIncomingAction(payload))
                    this.button.addEventListener('click', () => this.setColor(this.randomColor()))
                    await alleo.initialize()
                    console.log('[SyncedColorWidget.start] Widget initialized')
                }

                /**
                 * Rebuilds the UI from the received synced status.
                 * @param {Record<string, unknown>} [status] Latest full shared state.
                 */
                applyStatus(status) {
                    const color = status && status[this.statusKey] ? String(status[this.statusKey]) : this.defaultColor
                    this.swatch.style.setProperty('--swatch-color', color)
                    this.valueElement.textContent = color
                    console.log('[SyncedColorWidget.applyStatus]', color)
                }

                /**
                 * Handles incoming set-color and reset actions.
                 * @param {{ actionId?: string, data?: Record<string, unknown> }} payload Incoming payload.
                 */
                handleIncomingAction(payload) {
                    if (payload?.actionId === 'reset') {
                        this.setColor(this.defaultColor)
                        return
                    }
                    if (payload?.actionId === 'set-color') {
                        const color = payload.data?.color
                        if (typeof color === 'string' && color.trim()) this.setColor(color.trim())
                    }
                }

                /**
                 * Writes a color to the shared state and fires the change action.
                 * The subscription redraws the UI from the synced status.
                 * @param {string} color CSS color to store.
                 */
                setColor(color) {
                    alleo.setSyncedStatus({ [this.statusKey]: color })
                    alleo.triggerAction('color-changed', { color })
                    console.log('[SyncedColorWidget.setColor] wrote', color)
                }

                /**
                 * @returns {string} A random hex color string.
                 */
                randomColor() {
                    return `#${Math.floor(Math.random() * 0xffffff)
                        .toString(16)
                        .padStart(6, '0')}`
                }
            }

            /** @type {HTMLElement} Swatch element. */
            const swatch = document.getElementById('swatch')

            /** @type {HTMLElement} Color value element. */
            const valueElement = document.getElementById('value')

            /** @type {HTMLButtonElement} Randomize button. */
            const button = document.getElementById('randomize')

            /** @type {SyncedColorWidget} Active widget controller. */
            const syncedColorWidget = new SyncedColorWidget(swatch, valueElement, button)

            void syncedColorWidget.start()
        </script>
