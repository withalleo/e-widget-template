  board viewer sees the same value.

  Outgoing actions (triggered via alleo.triggerAction):
    - onloaded ()   fired once after init

  Incoming actions (handled via alleo.onIncomingAction):
    - none

  Synced status:
    - Writes { color: string } via alleo.setSyncedStatus on each click
    - Reads the full status via alleo.onSyncedStatusUpdate
    - Requests it on startup via alleo.requestSyncedStatus

  ⚠ REQUIRED SETTINGS: "Enable Alleo board features for enclosed content"
     and "Enable synchronizing status".
-->

<script>
    'use strict'

    /**
     * Stores a single shared color in synced status so every board viewer sees
     * the same value. The synced status is the source of truth for the UI.
     */
    class SyncedColorWidget {
        /* ═══════════════ Configuration ═══════════════ */

        /** @type {string} Key under which the color is stored in synced status. */
        statusKey = settings.statusKey

        /** @type {string} Fallback button color used before any status arrives. */
        defaultColor = settings.defaultColor

        /* ═══════════════ End Configuration ═══════════════ */

        /**
         * @param {object} alleo The Alleo SDK messenger (the global `alleo`).
         */
        constructor(alleo) {
            /** @type {object} Alleo SDK messenger. */
            this.alleo = alleo
            /** @type {HTMLButtonElement} Button whose background reflects the shared color. */
            this.button = document.getElementById('color')
        }

        /**
         * Subscribe to shared state, apply existing state, and announce readiness.
         * @fires onloaded
         */
        init() {
            this.alleo.onSyncedStatusUpdate((status) => this.applyStatus(status))
            this.button.addEventListener('click', () => this.setRandomColor())

            // Apply existing shared state before announcing readiness.
            this.alleo.requestSyncedStatus()
            this.alleo.triggerAction('onloaded')
            console.log('SyncedColorWidget.init: ready')
        }

        /**
         * Rebuild the UI from the received synced status.
         * @param {Record<string, unknown>=} status Latest full shared state.
         */
        applyStatus(status) {
            const color = status && status[this.statusKey] ? String(status[this.statusKey]) : this.defaultColor
            this.button.style.backgroundColor = color
            console.log('SyncedColorWidget.applyStatus:', color)
        }

        /**
         * Pick a random color, update the shared state, and let the subscription
         * redraw the button.
         */
        setRandomColor() {
            const color = `#${Math.floor(Math.random() * 0xffffff).toString(16).padStart(6, '0')}`
            this.alleo.setSyncedStatus({[this.statusKey]: color})
            console.log('SyncedColorWidget.setRandomColor: wrote', color)
        }
    }

    new SyncedColorWidget(alleo).init()
</script>
