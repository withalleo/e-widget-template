<!-- WIDGETHELP:
## What it does

This widget reads and lists all URL parameters that the page hosting Alleo shares
with the board. Only query parameters whose name starts with `__embed__` are
forwarded, and Alleo strips that prefix before exposing them through
`alleo.getParams`. Use it to verify what per-session values (for example a
`userId` or `lang`) reach your widgets.

## How to use it

1. Enable **Share "__embed__" URL parameters with enclosed content** in the
   widget's Security settings.
2. Add `__embed__`-prefixed query parameters to the URL of the page that embeds
   the Alleo board, e.g. `?__embed__userId=42&__embed__lang=en`.
3. Load the board. The widget lists each shared parameter and its value, and
   fires `params-listed` with the number of parameters found.

Send the `refresh-list` incoming action to re-render the list on demand.

## Settings

Board editors can adjust the background, primary, and text colors from the widget
settings. No configuration is required for the parameter lookup itself.

## Notes

Parameter values are captured once at startup and stay stable for the lifetime of
the embed. When the sharing setting is disabled, or no `__embed__` parameters are
present, the widget shows an empty-state message instead of a list.
-->

        <script>
            /**
             * Reads alleo.getParams and lists every shared __embed__ URL parameter.
             */
            class UrlParamsWidget {
                /* =============== Configuration =============== */

                /** @type {string} Heading shown above the parameter list. */
                heading = settings.heading

                /** @type {string} Message shown when no parameters are available. */
                emptyMessage = settings.emptyMessage

                /* =============== End Configuration =============== */

                /**
                 * Stores DOM references used by the controller.
                 * @param {Record<string, HTMLElement>} elements Named DOM elements.
                 */
                constructor(elements) {
                    /** @type {Record<string, HTMLElement>} Named DOM elements. */
                    this.el = elements
                }

                /**
                 * Registers incoming actions and initializes the SDK, then renders
                 * the parameters (available once initialize resolves).
                 */
                async start() {
                    this.el.heading.textContent = this.heading
                    this.el.empty.textContent = this.emptyMessage
                    alleo.onIncomingAction((payload) => this.handleIncomingAction(payload))
                    await alleo.initialize()
                    this.render()
                    console.log('[UrlParamsWidget.start] Widget initialized')
                }

                /**
                 * Re-renders the list when the refresh-list action is received.
                 * @param {{ actionId?: string }} payload Incoming payload.
                 */
                handleIncomingAction(payload) {
                    if (payload?.actionId === 'refresh-list') this.render()
                }

                /**
                 * Reads the shared parameters and renders them, then reports the count.
                 */
                render() {
                    const params = alleo.getParams || {}
                    const entries = Object.entries(params)

                    this.el.list.textContent = ''
                    this.el.count.textContent = entries.length === 1 ? '1 parameter' : `${entries.length} parameters`

                    const hasEntries = entries.length > 0
                    this.el.empty.classList.toggle('hidden', hasEntries)
                    this.el.list.classList.toggle('hidden', !hasEntries)

                    entries.forEach(([key, value]) => this.el.list.append(this.buildRow(key, value)))

                    alleo.triggerAction('params-listed', { count: entries.length })
                    console.log('[UrlParamsWidget.render] listed', entries.length, 'parameters')
                }

                /**
                 * Builds a key/value row.
                 * @param {string} key Parameter name (prefix already stripped).
                 * @param {string} value Parameter value.
                 * @returns {HTMLDivElement} The row element.
                 */
                buildRow(key, value) {
                    const row = document.createElement('div')
                    row.className = 'row'

                    const term = document.createElement('dt')
                    term.textContent = key

                    const description = document.createElement('dd')
                    description.textContent = String(value)

                    row.append(term, description)
                    return row
                }
            }

            /** @type {Record<string, HTMLElement>} Named DOM elements used by the widget. */
            const elements = {
                heading: document.getElementById('heading'),
                count: document.getElementById('count'),
                list: document.getElementById('list'),
                empty: document.getElementById('empty'),
            }

            /** @type {UrlParamsWidget} Active widget controller. */
            const urlParamsWidget = new UrlParamsWidget(elements)

            void urlParamsWidget.start()
        </script>
    </body>
</html>
