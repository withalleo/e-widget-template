<!-- WIDGETHELP:
## What it does

This widget is a minimal demonstration of the Alleo action system. It fires an
outgoing `demo` action when you tap the button, prints every incoming action it
receives, and shows the current shared synced status.

## How to use it

1. Import the widget and enable **Enable Alleo board features for enclosed
   content** (and **Enable synchronizing status** to see status updates).
2. Tap **Trigger demo action** to fire the `demo` action with a sample
   parameter; connect it to an automation to see it flow through the board.
3. Tap **Request synced status** to re-read the shared status on demand.
4. Send the widget a `demo-incoming` action from another board object to watch
   it appear in the on-screen log.

## Settings

Board editors can adjust the background, primary, and text colors. The action id,
parameter name, and sample value are grouped in the configuration section at the
top of the file.

## Notes

The widget only reads synced status; it never writes it. It is safe to run
without the synchronizing-status capability, in which case the status simply
stays empty.
-->

        <script>
            /**
             * Minimal demo widget: fires an outgoing action, logs incoming actions,
             * and reads the shared synced status.
             */
            class BasicActionsWidget {
                /* =============== Configuration =============== */

                /** @type {string} Outgoing action id fired by the trigger button. */
                triggerActionId = settings.triggerActionId

                /** @type {string} Parameter name carried by the trigger action. */
                triggerParamName = settings.triggerParamName

                /** @type {string} Payload value sent with the demo action. */
                triggerParamValue = settings.triggerParamValue

                /** @type {string} Incoming action id that is logged when received. */
                incomingActionId = settings.incomingActionId

                /** @type {number} Maximum number of lines kept in the on-screen log. */
                maxLogLines = settings.maxLogLines

                /* =============== End Configuration =============== */

                /**
                 * Stores DOM references used by the controller.
                 * @param {HTMLButtonElement} triggerButton Fires the outgoing action.
                 * @param {HTMLButtonElement} requestButton Requests the synced status.
                 * @param {HTMLElement} logElement On-screen log target.
                 */
                constructor(triggerButton, requestButton, logElement) {
                    /** @type {HTMLButtonElement} Fires the outgoing action. */
                    this.triggerButton = triggerButton
                    /** @type {HTMLButtonElement} Requests the synced status. */
                    this.requestButton = requestButton
                    /** @type {HTMLElement} On-screen log target. */
                    this.logElement = logElement
                }

                /**
                 * Registers listeners and integrations, then initializes the SDK.
                 */
                async start() {
                    this.bindEvents()
                    this.bindBoardHandlers()
                    await alleo.initialize()
                    this.log('Widget ready (onloaded fired).')
                    console.log('[BasicActionsWidget.start] Widget initialized')
                }

                /**
                 * Attaches touch/click handlers to the buttons.
                 */
                bindEvents() {
                    this.triggerButton.addEventListener('click', () => this.sendDemoAction())
                    this.requestButton.addEventListener('click', () => this.requestStatus())
                }

                /**
                 * Subscribes to synced-status updates and incoming actions.
                 */
                bindBoardHandlers() {
                    alleo.onSyncedStatusUpdate((status) => this.log('syncedStatusUpdate', status))
                    alleo.onIncomingAction(({ actionId, data }) => this.log(`incomingAction: ${actionId}`, data))
                }

                /**
                 * Fires the outgoing demo action with its configured payload.
                 */
                sendDemoAction() {
                    alleo.triggerAction(this.triggerActionId, { [this.triggerParamName]: this.triggerParamValue })
                    this.log(`triggerAction sent: ${this.triggerActionId}`)
                    console.log('[BasicActionsWidget.sendDemoAction] fired', this.triggerActionId)
                }

                /**
                 * Asks the board to resend the current synced status.
                 */
                requestStatus() {
                    alleo.requestSyncedStatus()
                    this.log('requestSyncedStatus sent')
                }

                /**
                 * Prepends a line to the on-screen log, trimmed to maxLogLines.
                 * @param {string} message Text to log.
                 * @param {unknown} [data] Optional value appended as JSON.
                 */
                log(message, data) {
                    const line = data !== undefined ? `${message} ${JSON.stringify(data)}` : message
                    this.logElement.textContent = `${line}\n${this.logElement.textContent}`
                        .split('\n')
                        .slice(0, this.maxLogLines)
                        .join('\n')
                }
            }

            /** @type {HTMLButtonElement} Fires the outgoing action. */
            const triggerButton = document.getElementById('trigger')

            /** @type {HTMLButtonElement} Requests the synced status. */
            const requestButton = document.getElementById('request')

            /** @type {HTMLElement} On-screen log target. */
            const logElement = document.getElementById('log')

            /** @type {BasicActionsWidget} Active widget controller. */
            const basicActionsWidget = new BasicActionsWidget(triggerButton, requestButton, logElement)

            void basicActionsWidget.start()
        </script>
