<!--
  WIDGET SUMMARY
  ==============

  An Alleo E-widget that captures microphone audio through the alleo.mic
  bridge and visualises it as a live waveform.

  Outgoing actions (triggered via alleo.triggerAction):
    - onloaded ()   fired once after init

  Incoming actions (handled via alleo.onIncomingAction):
    - none

  Microphone (via alleo.mic):
    - start({ timeslice }) / stop()
    - onTrack registered BEFORE start(); onStarted / onStopped / onError handled

  DEBUG: This sample logs verbose diagnostics to the on-screen log and the
  console to troubleshoot "no audio in the player":
    - track state (enabled / muted / readyState / settings) on onTrack
    - every <audio> element event (play, playing, stalled, error, volumechange…)
    - the monitor.play() promise result (autoplay-policy rejections are common)
    - AudioContext state + an explicit resume() (it often starts 'suspended')
    - a periodic input level (RMS) so you can tell if the stream is silent

  ⚠ WARNING: This widget requests microphone access via alleo.mic. The user must
             enable "Allow content access to the microphone". The parent prompts
             for permission on the first alleo.mic.start() call. (Camera access is
             not available inside the iframe.)
-->

<script>
    'use strict'

    /**
     * Captures microphone audio through the alleo.mic bridge, plays it back via
     * an <audio> element, and visualises it as a live waveform.
     *
     * Includes verbose debugging to diagnose "no audio in the player": the SDK
     * builds the stream from a hidden, MUTED <audio> + captureStream(), so the
     * delivered track can be silent, and the monitor element's autoplay can be
     * blocked. Every relevant state transition is logged.
     */
    class MicrophoneWidget {
        /* ═══════════════ Configuration ═══════════════ */

        /** @type {number} FFT window size for the analyser (power of two). */
        fftSize = 1024

        /** @type {string} Waveform stroke color. */
        waveColor = '#2c8'

        /** @type {number} Waveform stroke width in pixels. */
        waveLineWidth = 2

        /** @type {string} Canvas background color. */
        canvasBackground = '#111'

        /** @type {number} Default capture chunk size in ms (clamped to [min,max]Timeslice). */
        defaultTimeslice = 100

        /** @type {number} Minimum capture chunk size in ms. */
        minTimeslice = 20

        /** @type {number} Maximum capture chunk size in ms (hard SDK limit). */
        maxTimeslice = 200

        /** @type {number} How often (ms) to log the input level (RMS). */
        levelLogIntervalMs = 1000

        /** @type {number} Playback volume for the monitor element (0–1). */
        monitorVolume = 1.0

        /** @type {number} Maximum number of lines kept in the on-screen log. */
        maxLogLines = 300

        /** @type {string[]} <audio> events logged for debugging. */
        monitorEvents = [
            'loadedmetadata', 'play', 'playing', 'pause', 'waiting',
            'stalled', 'suspend', 'emptied', 'ended', 'error', 'volumechange',
        ]

        /* ═══════════════ End Configuration ═══════════════ */

        /**
         * @param {object} alleo The Alleo SDK messenger (the global `alleo`).
         */
        constructor(alleo) {
            /** @type {object} Alleo SDK messenger. */
            this.alleo = alleo
            /** @type {HTMLElement} On-screen log target. */
            this.logEl = document.getElementById('log')
            /** @type {HTMLButtonElement} Starts capture. */
            this.startBtn = document.getElementById('start')
            /** @type {HTMLButtonElement} Stops capture. */
            this.stopBtn = document.getElementById('stop')
            /** @type {HTMLAudioElement} Plays back the captured stream. */
            this.monitor = document.getElementById('monitor')
            /** @type {HTMLCanvasElement} Waveform canvas. */
            this.canvas = document.getElementById('scope')
            /** @type {CanvasRenderingContext2D} 2D drawing context. */
            this.ctx2d = this.canvas.getContext('2d')
            /** @type {HTMLElement} Live input-level readout. */
            this.levelEl = document.getElementById('level')
            /** @type {HTMLInputElement} Chunk-size input. */
            this.timesliceInput = document.getElementById('timeslice')

            /** @type {number} requestAnimationFrame handle for the visualiser. */
            this.analyserRaf = 0
            /** @type {AudioContext|null} Web Audio context for analysis. */
            this.audioCtx = null
            /** @type {number} Timestamp of the last level log. */
            this.lastLevelLog = 0
        }

        /**
         * Register mic handlers (BEFORE start), wire buttons, and announce readiness.
         * @fires onloaded
         */
        init() {
            this.registerMonitorDebugEvents()
            this.alleo.mic.onStarted((info) => this.handleStarted(info))
            this.alleo.mic.onStopped(() => this.handleStopped())
            this.alleo.mic.onError(({error}) => this.handleMicError(error))
            this.alleo.mic.onTrack((stream) => this.handleTrack(stream))

            this.startBtn.addEventListener('click', () => this.startCapture())
            this.stopBtn.addEventListener('click', () => this.stopCapture())

            // Mic handlers registered and listeners attached — announce readiness.
            this.alleo.triggerAction('onloaded')
            console.log('MicrophoneWidget.init: ready')
        }

        /** Log every meaningful <audio> element event to aid debugging. */
        registerMonitorDebugEvents() {
            for (const evt of this.monitorEvents) {
                this.monitor.addEventListener(evt, () => this.log(`audio event: ${evt}`, {
                    readyState: this.monitor.readyState,
                    paused: this.monitor.paused,
                    muted: this.monitor.muted,
                    volume: this.monitor.volume,
                    currentTime: Number(this.monitor.currentTime.toFixed(2)),
                }))
            }
        }

        /**
         * Start microphone capture with the chosen (clamped) chunk size.
         */
        startCapture() {
            const timeslice = Math.min(
                this.maxTimeslice,
                Math.max(this.minTimeslice, Number(this.timesliceInput.value) || this.defaultTimeslice),
            )
            this.alleo.mic.start({timeslice})
            this.log(`mic.start({ timeslice: ${timeslice} })`)
            console.log('MicrophoneWidget.startCapture:', timeslice)
        }

        /** Stop microphone capture. */
        stopCapture() {
            this.alleo.mic.stop()
            this.log('mic.stop()')
        }

        /**
         * @param {{mimeType?: string, sampleRate?: number, timeslice?: number}=} info Capture info.
         */
        handleStarted(info) {
            this.log('mic started', info)
            this.startBtn.disabled = true
            this.stopBtn.disabled = false
        }

        /** Tear down the visualiser and detach the monitor when capture stops. */
        handleStopped() {
            this.log('mic stopped')
            this.startBtn.disabled = false
            this.stopBtn.disabled = true
            cancelAnimationFrame(this.analyserRaf)
            this.monitor.srcObject = null
            this.levelEl.textContent = 'Input level: —'
            if (this.audioCtx) {
                this.audioCtx.close().catch(() => {})
                this.audioCtx = null
            }
        }

        /**
         * @param {string} error Error description from the mic bridge.
         */
        handleMicError(error) {
            this.log('mic error: ' + error)
            console.error('MicrophoneWidget.handleMicError:', error)
            this.startBtn.disabled = false
            this.stopBtn.disabled = true
        }

        /**
         * Receive the live MediaStream, start playback, and start the visualiser.
         *
         * The SDK builds the stream from a hidden <audio> + captureStream(), so the
         * audio track is added asynchronously once the first chunk decodes — the
         * stream therefore usually arrives with zero tracks. We attach it to the
         * monitor right away (the <audio> element starts playing as soon as the
         * track appears) and (re)build the visualiser via the stream's `addtrack`
         * event so a late-arriving track is not missed.
         * @param {MediaStream} stream Live audio stream from the parent.
         */
        handleTrack(stream) {
            this.logTracks(stream)
            this.monitor.srcObject = stream
            this.monitor.muted = false
            this.monitor.volume = this.monitorVolume

            if (stream.getAudioTracks().length > 0) this.onStreamReady(stream)

            stream.addEventListener('addtrack', (event) => {
                this.log('audio track arrived', {kind: event.track.kind})
                this.onStreamReady(stream)
            })
        }

        /**
         * Start playback and the visualiser once the stream actually carries audio.
         * Guarded so it runs at most once per capture session.
         * @param {MediaStream} stream Stream that now has an audio track.
         */
        onStreamReady(stream) {
            this.playMonitor()
            if (!this.audioCtx) this.startVisualizer(stream)
        }

        /**
         * Log the audio-track state of a stream for debugging.
         * @param {MediaStream} stream Stream to inspect.
         */
        logTracks(stream) {
            const tracks = stream.getAudioTracks()
            this.log('onTrack: stream received', {tracks: tracks.length})
            tracks.forEach((track, i) => this.log(`track[${i}] debug`, {
                enabled: track.enabled,
                muted: track.muted,
                readyState: track.readyState,
                label: track.label,
                settings: typeof track.getSettings === 'function' ? track.getSettings() : null,
            }))
            if (tracks.length === 0) this.log('stream has no audio tracks yet — waiting for addtrack')
        }

        /**
         * Play the monitor element once it has data. An AbortError means a newer
         * load (a fresh onTrack/srcObject) superseded this attempt — that is
         * expected and benign, so it is logged at info level, not as a warning.
         */
        playMonitor() {
            const attempt = () => {
                const playback = this.monitor.play()
                if (!playback || typeof playback.then !== 'function') return
                playback
                    .then(() => this.log('monitor.play() resolved — element is playing'))
                    .catch((err) => {
                        if (err.name === 'AbortError') {
                            this.log('monitor.play() superseded by a newer load (benign)')
                            return
                        }
                        this.log('monitor.play() REJECTED (autoplay blocked?)', {name: err.name, message: err.message})
                        console.warn('MicrophoneWidget.playMonitor: play() rejected', err)
                    })
            }
            // HAVE_CURRENT_DATA (2) or better means we can start without racing a load.
            if (this.monitor.readyState >= 2) attempt()
            else this.monitor.addEventListener('canplay', attempt, {once: true})
        }

        /**
         * Build the Web Audio graph and start the waveform/level loop. Resumes a
         * suspended AudioContext (common under autoplay policies).
         * @param {MediaStream} stream Stream to analyse.
         */
        startVisualizer(stream) {
            this.audioCtx = new (window.AudioContext || window.webkitAudioContext)()
            this.log('AudioContext created', {state: this.audioCtx.state, sampleRate: this.audioCtx.sampleRate})
            if (this.audioCtx.state === 'suspended') {
                this.audioCtx.resume()
                    .then(() => this.log('AudioContext resumed', {state: this.audioCtx.state}))
                    .catch((err) => this.log('AudioContext.resume() failed', {message: err.message}))
            }

            const source = this.audioCtx.createMediaStreamSource(stream)
            const analyser = this.audioCtx.createAnalyser()
            analyser.fftSize = this.fftSize
            source.connect(analyser)
            const buffer = new Uint8Array(analyser.fftSize)

            const draw = () => {
                this.analyserRaf = requestAnimationFrame(draw)
                analyser.getByteTimeDomainData(buffer)
                this.renderWave(buffer)
                this.reportLevel(buffer)
            }
            draw()
        }

        /**
         * Draw one frame of the time-domain waveform.
         * @param {Uint8Array} buffer Time-domain samples (0–255, 128 = silence).
         */
        renderWave(buffer) {
            const {ctx2d, canvas} = this
            ctx2d.fillStyle = this.canvasBackground
            ctx2d.fillRect(0, 0, canvas.width, canvas.height)
            ctx2d.lineWidth = this.waveLineWidth
            ctx2d.strokeStyle = this.waveColor
            ctx2d.beginPath()
            const slice = canvas.width / buffer.length
            for (let i = 0; i < buffer.length; i++) {
                const v = buffer[i] / 128.0
                const y = (v * canvas.height) / 2
                const x = i * slice
                if (i === 0) ctx2d.moveTo(x, y)
                else ctx2d.lineTo(x, y)
            }
            ctx2d.lineTo(canvas.width, canvas.height / 2)
            ctx2d.stroke()
        }

        /**
         * Compute the RMS level and update the readout; periodically log it so a
         * silent (muted-source) stream is easy to spot.
         * @param {Uint8Array} buffer Time-domain samples (0–255, 128 = silence).
         */
        reportLevel(buffer) {
            let sumSquares = 0
            for (let i = 0; i < buffer.length; i++) {
                const v = (buffer[i] - 128) / 128
                sumSquares += v * v
            }
            const rms = Math.sqrt(sumSquares / buffer.length)
            this.levelEl.textContent = `Input level (RMS): ${rms.toFixed(4)}`

            const now = performance.now()
            if (now - this.lastLevelLog < this.levelLogIntervalMs) return
            this.lastLevelLog = now
            this.log('input level (RMS)', {rms: Number(rms.toFixed(4))})
            if (rms === 0) {
                this.log('⚠ level is exactly 0 — the delivered stream is silent (muted source / capture issue)')
            }
        }

        /**
         * Prepend a line to the on-screen log, trimmed to {@link maxLogLines}.
         * @param {string} message Text to log.
         * @param {unknown=} data Optional value JSON-stringified after the message.
         */
        log(message, data) {
            const line = data !== undefined ? `${message} ${JSON.stringify(data)}` : message
            this.logEl.textContent = `${line}\n${this.logEl.textContent}`
                .split('\n').slice(0, this.maxLogLines).join('\n')
        }
    }

    new MicrophoneWidget(alleo).init()
</script>
