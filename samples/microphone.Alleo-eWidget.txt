<!-- WIDGETHELP:
## What it does

This widget demonstrates microphone capture from inside a sandboxed E-widget.
Because `getUserMedia()` is blocked, it uses the `alleo.mic` bridge: the parent
Alleo widget records the microphone and streams the audio back into the iframe as
a live `MediaStream`, which the widget visualises as a waveform and an input
level meter.

## How to use it

1. Enable **Allow content access to the microphone** in the widget settings.
2. Tap **Start** and grant the browser microphone permission when prompted on the
   parent page.
3. Speak to see the live waveform and RMS level respond.
4. Tap **Stop** to release the microphone.

The widget can also be controlled from the board with the `start-mic` and
`stop-mic` incoming actions, and it emits `mic-started`, `mic-stopped`, and
`mic-error` outgoing actions.

## Settings

Board editors can adjust the background, primary, and text colors. The waveform
resolution and chunk size live in the configuration section at the top of the
file.

## Notes

Microphone access requires the **Allow content access to the microphone** widget
setting. Camera access is not available inside the iframe. Always stop capture
when it is no longer needed so the parent releases the device.
-->

        <script>
            /**
             * Captures microphone audio through the alleo.mic bridge and renders a
             * live waveform and input-level meter.
             */
            class MicrophoneWidget {
                /* =============== Configuration =============== */

                /** @type {number} Capture chunk size in ms. */
                timeslice = settings.timeslice

                /** @type {number} FFT window size for the analyser. */
                fftSize = settings.fftSize

                /** @type {number} Waveform stroke width in pixels. */
                waveLineWidth = settings.waveLineWidth

                /* =============== End Configuration =============== */

                /**
                 * Stores DOM references and analysis state.
                 * @param {Record<string, HTMLElement>} elements Named DOM elements.
                 */
                constructor(elements) {
                    /** @type {Record<string, HTMLElement>} Named DOM elements. */
                    this.el = elements
                    /** @type {CanvasRenderingContext2D} 2D drawing context for the waveform. */
                    this.ctx = elements.scope.getContext('2d')
                    /** @type {AudioContext | null} Web Audio context for analysis. */
                    this.audioCtx = null
                    /** @type {number} requestAnimationFrame handle for the visualiser. */
                    this.rafHandle = 0
                }

                /**
                 * Registers mic handlers (before start), wires buttons, and
                 * initializes the SDK.
                 */
                async start() {
                    // onTrack must be registered BEFORE start().
                    alleo.mic.onTrack((stream) => this.handleTrack(stream))
                    alleo.mic.onStarted((info) => this.handleStarted(info))
                    alleo.mic.onStopped(() => this.handleStopped())
                    alleo.mic.onError(({ error }) => this.handleMicError(error))

                    this.el.start.addEventListener('click', () => this.startCapture())
                    this.el.stop.addEventListener('click', () => this.stopCapture())
                    alleo.onIncomingAction((payload) => this.handleIncomingAction(payload))

                    await alleo.initialize()
                    console.log('[MicrophoneWidget.start] Widget initialized')
                }

                /**
                 * Asks the parent to start microphone capture.
                 */
                startCapture() {
                    alleo.mic.start({ timeslice: this.timeslice })
                    this.el.status.textContent = 'Requesting microphone…'
                    console.log('[MicrophoneWidget.startCapture]', this.timeslice)
                }

                /**
                 * Asks the parent to stop microphone capture.
                 */
                stopCapture() {
                    alleo.mic.stop()
                }

                /**
                 * Handles incoming start-mic / stop-mic actions.
                 * @param {{ actionId?: string }} payload Incoming payload.
                 */
                handleIncomingAction(payload) {
                    if (payload?.actionId === 'start-mic') this.startCapture()
                    if (payload?.actionId === 'stop-mic') this.stopCapture()
                }

                /**
                 * Reflects the started state and reports it to the board.
                 * @param {{ mimeType?: string, sampleRate?: number, timeslice?: number }} [info] Capture info.
                 */
                handleStarted(info) {
                    this.el.start.disabled = true
                    this.el.stop.disabled = false
                    this.el.status.textContent = 'Listening…'
                    alleo.triggerAction('mic-started', {})
                    console.log('[MicrophoneWidget.handleStarted]', info)
                }

                /**
                 * Tears down analysis and reports the stopped state.
                 */
                handleStopped() {
                    this.el.start.disabled = false
                    this.el.stop.disabled = true
                    this.el.status.textContent = 'Stopped.'
                    this.el.meter.style.width = '0%'
                    cancelAnimationFrame(this.rafHandle)
                    if (this.audioCtx) {
                        this.audioCtx.close().catch(() => {})
                        this.audioCtx = null
                    }
                    alleo.triggerAction('mic-stopped', {})
                }

                /**
                 * Reports a microphone error to the UI and the board.
                 * @param {string} error Error description from the mic bridge.
                 */
                handleMicError(error) {
                    this.el.start.disabled = false
                    this.el.stop.disabled = true
                    this.el.status.textContent = `Microphone error: ${error}`
                    alleo.triggerAction('mic-error', { error: String(error) })
                    console.error('[MicrophoneWidget.handleMicError]', error)
                }

                /**
                 * Starts the visualiser once the live stream carries an audio track.
                 * @param {MediaStream} stream Live audio stream from the parent.
                 */
                handleTrack(stream) {
                    if (stream.getAudioTracks().length > 0) this.startVisualizer(stream)
                    stream.addEventListener('addtrack', () => this.startVisualizer(stream))
                }

                /**
                 * Builds the Web Audio graph and starts the waveform/level loop.
                 * @param {MediaStream} stream Stream to analyse.
                 */
                startVisualizer(stream) {
                    if (this.audioCtx) return
                    this.audioCtx = new (window.AudioContext || window.webkitAudioContext)()
                    if (this.audioCtx.state === 'suspended') this.audioCtx.resume().catch(() => {})

                    const source = this.audioCtx.createMediaStreamSource(stream)
                    const analyser = this.audioCtx.createAnalyser()
                    analyser.fftSize = this.fftSize
                    source.connect(analyser)
                    const buffer = new Uint8Array(analyser.fftSize)

                    const draw = () => {
                        this.rafHandle = requestAnimationFrame(draw)
                        analyser.getByteTimeDomainData(buffer)
                        this.renderWave(buffer)
                        this.renderLevel(buffer)
                    }
                    draw()
                }

                /**
                 * Draws one frame of the time-domain waveform.
                 * @param {Uint8Array} buffer Time-domain samples (0–255, 128 = silence).
                 */
                renderWave(buffer) {
                    const { ctx } = this
                    const { width, height } = this.el.scope
                    ctx.clearRect(0, 0, width, height)
                    ctx.lineWidth = this.waveLineWidth
                    ctx.strokeStyle = getComputedStyle(document.body).getPropertyValue('--alleo-primary-color') || '#6da8ff'
                    ctx.beginPath()
                    const slice = width / buffer.length
                    for (let i = 0; i < buffer.length; i++) {
                        const y = (buffer[i] / 255) * height
                        const x = i * slice
                        if (i === 0) ctx.moveTo(x, y)
                        else ctx.lineTo(x, y)
                    }
                    ctx.stroke()
                }

                /**
                 * Computes the RMS level and updates the meter width.
                 * @param {Uint8Array} buffer Time-domain samples (0–255, 128 = silence).
                 */
                renderLevel(buffer) {
                    let sumSquares = 0
                    for (let i = 0; i < buffer.length; i++) {
                        const v = (buffer[i] - 128) / 128
                        sumSquares += v * v
                    }
                    const rms = Math.sqrt(sumSquares / buffer.length)
                    this.el.meter.style.width = `${Math.min(100, Math.round(rms * 300))}%`
                }
            }

            /** @type {Record<string, HTMLElement>} Named DOM elements used by the widget. */
            const elements = {
                scope: document.getElementById('scope'),
                meter: document.getElementById('meter'),
                start: document.getElementById('start'),
                stop: document.getElementById('stop'),
                status: document.getElementById('status'),
            }

            /** @type {MicrophoneWidget} Active widget controller. */
            const microphoneWidget = new MicrophoneWidget(elements)

            void microphoneWidget.start()
        </script>
    </body>
</html>
