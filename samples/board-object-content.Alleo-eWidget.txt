  WIDGET SUMMARY
  ==============

  An Alleo E-widget that reads and writes the text of a whitelisted board
  object (get / replace / append).

  Outgoing actions (triggered via alleo.triggerAction):
    - onloaded ()   fired once after init

  Incoming actions (handled via alleo.onIncomingAction):
    - none

  Board object content (whitelist index N chosen by the user):
    - Reads object at index N            via alleo.getBoardObjectContent
    - Writes (replace) object at index N via alleo.replaceBoardObjectContent
    - Writes (append) object at index N  via alleo.appendBoardObjectContent
    - Errors handled via alleo.onError

  ⚠ REQUIRED SETTINGS: "Enable Alleo board features for enclosed content",
     "Enable reading/writing board object content", and at least one object
     added to "Whitelisted board objects" in widget settings.
-->

<script>
    'use strict'

    /**
     * Reads and writes the text of a whitelisted board object: get / replace /
     * append, plus a global error handler.
     */
    class BoardObjectContentWidget {
        /* ═══════════════ Configuration ═══════════════ */

        /** @type {number} Fallback whitelist index (1-based) when the input is empty/invalid. */
        defaultIndex = 1

        /** @type {number} Maximum number of lines kept in the on-screen log. */
        maxLogLines = 200

        /* ═══════════════ End Configuration ═══════════════ */

        /**
         * @param {object} alleo The Alleo SDK messenger (the global `alleo`).
         */
        constructor(alleo) {
            /** @type {object} Alleo SDK messenger. */
            this.alleo = alleo
            /** @type {HTMLElement} On-screen log target. */
            this.logEl = document.getElementById('log')
            /** @type {HTMLInputElement} Whitelist index input (1-based). */
            this.indexInput = document.getElementById('index')
            /** @type {HTMLSelectElement} Content format selector. */
            this.formatSelect = document.getElementById('format')
            /** @type {HTMLTextAreaElement} Text used for replace/append and shown on get. */
            this.textArea = document.getElementById('text')
        }

        /**
         * Wire buttons, register the error handler, and announce readiness.
         * @fires onloaded
         */
        init() {
            document.getElementById('get').addEventListener('click', () => this.getContent())
            document.getElementById('replace').addEventListener('click', () => this.replaceContent())
            document.getElementById('append').addEventListener('click', () => this.appendContent())
            this.alleo.onError(({index, error, requestId}) => this.handleError(index, error, requestId))

            // Listeners attached — announce the widget is ready.
            this.alleo.triggerAction('onloaded')
            console.log('BoardObjectContentWidget.init: ready')
        }

        /** @returns {number} The selected whitelist index (1-based), or the default. */
        get index() {
            return Number(this.indexInput.value) || this.defaultIndex
        }

        /** @returns {string} The selected content format ('text' | 'markdown' | 'html'). */
        get format() {
            return this.formatSelect.value
        }

        /** Read the whitelisted object's text and show it in the textarea. */
        async getContent() {
            const {index, format} = this
            this.log(`getBoardObjectContent(${index}, { format: '${format}' }) …`)
            try {
                const content = await this.alleo.getBoardObjectContent(index, {format})
                this.log('Content received:', content)
                this.textArea.value = content.join('\n')
            } catch (err) {
                this.log('Error: ' + err.message)
                console.warn('BoardObjectContentWidget.getContent failed:', err)
            }
        }

        /** Replace the whitelisted object's text with the textarea contents. */
        replaceContent() {
            const {index, format} = this
            this.alleo.replaceBoardObjectContent(index, this.textArea.value, {format})
            this.log(`replaceBoardObjectContent(${index}) sent`)
        }

        /** Append the textarea contents to the whitelisted object's text. */
        appendContent() {
            const {index, format} = this
            this.alleo.appendBoardObjectContent(index, this.textArea.value, {format})
            this.log(`appendBoardObjectContent(${index}) sent`)
        }

        /**
         * Log a board-object error reported by the SDK.
         * @param {number} index Whitelist index that failed.
         * @param {string} error Error message.
         * @param {string=} requestId Correlation id of the failed request.
         */
        handleError(index, error, requestId) {
            this.log(`boardObjectContentError #${index} (requestId=${requestId}):`, error)
            console.error('BoardObjectContentWidget.handleError:', index, error)
        }

        /**
         * Prepend a line to the on-screen log, trimmed to {@link maxLogLines}.
         * @param {string} message Text to log.
         * @param {unknown=} data Optional value JSON-stringified after the message.
         */
        log(message, data) {
            const line = data !== undefined ? `${message} ${JSON.stringify(data)}` : message
            this.logEl.textContent = `${line}\n${this.logEl.textContent}`
                .split('\n').slice(0, this.maxLogLines).join('\n')
        }
    }

    new BoardObjectContentWidget(alleo).init()
</script>
