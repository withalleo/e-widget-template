Reads, replaces, and appends the text of a whitelisted board object using the Alleo board-object content API.
-->
<!-- WIDGETHELP:
## What it does

This widget demonstrates reading and writing the text content of other board
objects. Pick a whitelist index and a text format, then **Get**, **Replace**, or
**Append** the target object's content. Results and errors are shown in an
on-screen log.

## How to use it

1. Enable **Enable Alleo board features**, **Enable reading/writing board object
   content**, and add one or more objects to the **Whitelisted board objects**
   list in the widget settings.
2. Set **Index** to the 1-based position of the object in that whitelist.
3. Choose a **Format** (text, markdown, or html) from the custom dropdown.
4. Tap **Get** to load the object's text into the box, or type text and tap
   **Replace** or **Append** to write it back.

The `read-object` incoming action can trigger a read from another board object;
its optional `index` parameter overrides the on-screen index.

## Settings

Board editors can adjust the background, primary, and text colors. The default
index and format live in the configuration section near the top of the file.

## Notes

Write operations are fire-and-forget; failures are reported through the log via
`alleo.onError`. The whitelist indices are **1-based positions**, not object ids.
-->

        <script>
            /**
             * Reads and writes the text of a whitelisted board object (get / replace /
             * append) and reports outcomes through the log and outgoing actions.
             */
            class BoardObjectContentWidget {
                /* =============== Configuration =============== */

                /** @type {number} Fallback whitelist index (1-based). */
                defaultIndex = settings.defaultIndex

                /** @type {string[]} Selectable content formats. */
                formats = settings.formats

                /** @type {string} Currently selected format. */
                format = settings.defaultFormat

                /** @type {number} Maximum number of lines kept in the log. */
                maxLogLines = settings.maxLogLines

                /* =============== End Configuration =============== */

                /**
                 * Stores DOM references used by the controller.
                 * @param {Record<string, HTMLElement>} elements Named DOM elements.
                 */
                constructor(elements) {
                    /** @type {Record<string, HTMLElement>} Named DOM elements. */
                    this.el = elements
                }

                /**
                 * Builds the dropdown, wires listeners, registers error handling,
                 * and initializes the SDK.
                 */
                async start() {
                    this.buildFormatOptions()
                    this.bindEvents()
                    alleo.onError(({ index, error, requestId }) => this.handleError(index, error, requestId))
                    alleo.onIncomingAction((payload) => this.handleIncomingAction(payload))
                    await alleo.initialize()
                    console.log('[BoardObjectContentWidget.start] Widget initialized')
                }

                /**
                 * Renders the custom format dropdown options.
                 */
                buildFormatOptions() {
                    this.el.formatList.textContent = ''
                    this.formats.forEach((format) => {
                        const option = document.createElement('li')
                        option.className = 'dropdown__option'
                        option.role = 'option'
                        option.textContent = format
                        option.dataset.value = format
                        option.setAttribute('aria-selected', String(format === this.format))
                        option.addEventListener('click', () => this.selectFormat(format))
                        this.el.formatList.append(option)
                    })
                }

                /**
                 * Attaches touch/click handlers for the dropdown and action buttons.
                 */
                bindEvents() {
                    this.el.formatButton.addEventListener('click', () => this.toggleDropdown())
                    document.addEventListener('click', (event) => {
                        if (!this.el.formatDropdown.contains(event.target)) this.setDropdownOpen(false)
                    })
                    this.el.get.addEventListener('click', () => this.getContent())
                    this.el.replace.addEventListener('click', () => this.writeContent('replace'))
                    this.el.append.addEventListener('click', () => this.writeContent('append'))
                }

                /**
                 * Toggles the custom dropdown open/closed.
                 */
                toggleDropdown() {
                    this.setDropdownOpen(this.el.formatDropdown.dataset.open !== 'true')
                }

                /**
                 * Opens or closes the dropdown.
                 * @param {boolean} isOpen Whether the dropdown should be open.
                 */
                setDropdownOpen(isOpen) {
                    this.el.formatDropdown.dataset.open = String(isOpen)
                    this.el.formatButton.setAttribute('aria-expanded', String(isOpen))
                }

                /**
                 * Selects a content format and updates the dropdown UI.
                 * @param {string} format Chosen format.
                 */
                selectFormat(format) {
                    this.format = format
                    this.el.formatValue.textContent = format
                    this.setDropdownOpen(false)
                    this.el.formatList.querySelectorAll('.dropdown__option').forEach((option) => {
                        option.setAttribute('aria-selected', String(option.dataset.value === format))
                    })
                    console.log('[BoardObjectContentWidget.selectFormat]', format)
                }

                /** @returns {number} The selected whitelist index (1-based), or the default. */
                get index() {
                    return Number(this.el.index.value) || this.defaultIndex
                }

                /**
                 * Reads the whitelisted object's text and shows it in the textarea.
                 */
                async getContent() {
                    const { index, format } = this
                    this.log(`getBoardObjectContent(${index}, { format: '${format}' })…`)
                    try {
                        const content = await alleo.getBoardObjectContent(index, { format })
                        this.el.text.value = content.join('\n')
                        this.log('content received', content)
                        alleo.triggerAction('object-read', { index })
                    } catch (error) {
                        this.log('error: ' + (error?.message ?? error))
                    }
                }

                /**
                 * Replaces or appends the whitelisted object's text.
                 * @param {'replace' | 'append'} mode Write mode.
                 */
                writeContent(mode) {
                    const { index, format } = this
                    const text = this.el.text.value
                    if (mode === 'replace') {
                        alleo.replaceBoardObjectContent(index, text, { format })
                        alleo.triggerAction('object-replaced', { index })
                    } else {
                        alleo.appendBoardObjectContent(index, text, { format })
                        alleo.triggerAction('object-appended', { index })
                    }
                    this.log(`${mode}BoardObjectContent(${index}) sent`)
                }

                /**
                 * Handles the incoming read-object action.
                 * @param {{ actionId?: string, data?: Record<string, unknown> }} payload Incoming payload.
                 */
                handleIncomingAction(payload) {
                    if (payload?.actionId !== 'read-object') return
                    const index = Number(payload.data?.index)
                    if (Number.isFinite(index) && index > 0) this.el.index.value = String(index)
                    void this.getContent()
                }

                /**
                 * Logs a board-object error reported by the SDK.
                 * @param {number} [index] Whitelist index that failed.
                 * @param {string} [error] Error message.
                 * @param {string} [requestId] Correlation id of the failed request.
                 */
                handleError(index, error, requestId) {
                    this.log(`boardObjectError #${index} (requestId=${requestId})`, error)
                    console.error('[BoardObjectContentWidget.handleError]', index, error)
                }

                /**
                 * Prepends a line to the on-screen log, trimmed to maxLogLines.
                 * @param {string} message Text to log.
                 * @param {unknown} [data] Optional value appended as JSON.
                 */
                log(message, data) {
                    const line = data !== undefined ? `${message} ${JSON.stringify(data)}` : message
                    this.el.log.textContent = `${line}\n${this.el.log.textContent}`
                        .split('\n')
                        .slice(0, this.maxLogLines)
                        .join('\n')
                }
            }

            /** @type {Record<string, HTMLElement>} Named DOM elements used by the widget. */
            const elements = {
                index: document.getElementById('index'),
                formatDropdown: document.getElementById('format-dropdown'),
                formatButton: document.getElementById('format-button'),
                formatValue: document.getElementById('format-value'),
                formatList: document.getElementById('format-list'),
                get: document.getElementById('get'),
                replace: document.getElementById('replace'),
                append: document.getElementById('append'),
                text: document.getElementById('text'),
                log: document.getElementById('log'),
            }

            /** @type {BoardObjectContentWidget} Active widget controller. */
            const boardObjectContentWidget = new BoardObjectContentWidget(elements)

            void boardObjectContentWidget.start()
        </script>
