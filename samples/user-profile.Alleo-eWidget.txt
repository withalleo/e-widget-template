<!-- WIDGETHELP:
## What it does

This widget reads `alleo.user` and displays all available profile data for the
currently logged-in board user: name fields, email, organization, activity and
login times, and the user's display color. It is a handy reference for what user
information an E-widget can personalise itself with.

## How to use it

1. Enable **Allow access to the current user profile** in the widget's Security
   settings.
2. Load the board while signed in. The widget renders an avatar built from the
   user's initials and display color, followed by every populated profile field.
3. When the profile loads it fires `profile-loaded` with the display name and
   email so board automations can react.

Send the `refresh-profile` incoming action to re-render the profile on demand.

## Settings

Board editors can adjust the background, primary, and text colors from the widget
settings. No configuration is required for the profile lookup itself.

## Notes

The profile is captured once at startup and stays stable for the lifetime of the
embed. Fields the user has not set are omitted. To respect the sandbox rule
against external images, the user's `imageUrl` is shown as text rather than
loaded as a picture; the avatar is drawn from the initials and display color.
-->

        <script>
            /**
             * Reads alleo.user and displays every available profile field for the
             * current board user.
             */
            class UserProfileWidget {
                /* =============== Configuration =============== */

                /** @type {string} Message shown when no profile data is available. */
                emptyMessage = settings.emptyMessage

                /** @type {Array<{ key: string, label: string, type?: 'date' }>} Fields to display. */
                fields = settings.fields

                /* =============== End Configuration =============== */

                /**
                 * Stores DOM references used by the controller.
                 * @param {Record<string, HTMLElement>} elements Named DOM elements.
                 */
                constructor(elements) {
                    /** @type {Record<string, HTMLElement>} Named DOM elements. */
                    this.el = elements
                }

                /**
                 * Registers incoming actions and initializes the SDK, then renders
                 * the profile (available once initialize resolves).
                 */
                async start() {
                    this.el.empty.textContent = this.emptyMessage
                    alleo.onIncomingAction((payload) => this.handleIncomingAction(payload))
                    await alleo.initialize()
                    this.render()
                    console.log('[UserProfileWidget.start] Widget initialized')
                }

                /**
                 * Re-renders the profile when the refresh-profile action is received.
                 * @param {{ actionId?: string }} payload Incoming payload.
                 */
                handleIncomingAction(payload) {
                    if (payload?.actionId === 'refresh-profile') this.render()
                }

                /**
                 * Reads the user profile and renders the identity header and fields.
                 */
                render() {
                    const user = alleo.user || {}
                    const populated = this.fields
                        .map((field) => ({ field, value: this.readValue(user, field.key) }))
                        .filter((entry) => entry.value !== undefined && entry.value !== null && entry.value !== '')

                    const hasData = populated.length > 0
                    this.el.empty.classList.toggle('hidden', hasData)
                    this.el.identity.classList.toggle('hidden', !hasData)
                    this.el.fields.classList.toggle('hidden', !hasData)

                    if (hasData) {
                        this.renderIdentity(user)
                        this.renderFields(populated)
                    }

                    alleo.triggerAction('profile-loaded', {
                        displayName: this.resolveDisplayName(user),
                        email: typeof user.email === 'string' ? user.email : '',
                    })
                    console.log('[UserProfileWidget.render] rendered', populated.length, 'fields')
                }

                /**
                 * Renders the avatar, name, and email subtitle.
                 * @param {Record<string, any>} user The user profile object.
                 */
                renderIdentity(user) {
                    const displayName = this.resolveDisplayName(user)
                    this.el.name.textContent = displayName || 'Current user'
                    this.el.subtitle.textContent = typeof user.email === 'string' ? user.email : ''
                    this.el.avatar.textContent = this.resolveInitials(user)
                    if (typeof user.displayColor === 'string' && user.displayColor.trim()) {
                        this.el.avatar.style.setProperty('--avatar-color', user.displayColor.trim())
                    }
                }

                /**
                 * Renders one row per populated profile field.
                 * @param {Array<{ field: { label: string, key: string, type?: 'date' }, value: any }>} populated Populated fields.
                 */
                renderFields(populated) {
                    this.el.fields.textContent = ''
                    populated.forEach(({ field, value }) => {
                        const row = document.createElement('div')
                        row.className = 'row'

                        const term = document.createElement('dt')
                        term.textContent = field.label

                        const description = document.createElement('dd')
                        this.fillValue(description, field, value)

                        row.append(term, description)
                        this.el.fields.append(row)
                    })
                }

                /**
                 * Writes a formatted value into a description cell.
                 * @param {HTMLElement} cell Target cell.
                 * @param {{ key: string, type?: 'date' }} field Field descriptor.
                 * @param {any} value Raw field value.
                 */
                fillValue(cell, field, value) {
                    if (field.type === 'date') {
                        cell.textContent = this.formatDate(value)
                        return
                    }
                    if (field.key === 'displayColor') {
                        const swatch = document.createElement('span')
                        swatch.className = 'swatch'
                        swatch.style.background = String(value)
                        cell.append(swatch, document.createTextNode(String(value)))
                        return
                    }
                    cell.textContent = String(value)
                }

                /**
                 * Reads a possibly-nested value using dot notation.
                 * @param {Record<string, any>} source Object to read from.
                 * @param {string} path Dot-separated path (e.g. "organization.name").
                 * @returns {any} The resolved value, or undefined.
                 */
                readValue(source, path) {
                    return path.split('.').reduce((value, key) => (value == null ? value : value[key]), source)
                }

                /**
                 * Resolves the best available display name.
                 * @param {Record<string, any>} user The user profile object.
                 * @returns {string} A display name, possibly empty.
                 */
                resolveDisplayName(user) {
                    if (typeof user.displayName === 'string' && user.displayName.trim()) return user.displayName.trim()
                    return [user.firstName, user.lastName].filter((part) => typeof part === 'string' && part.trim()).join(' ')
                }

                /**
                 * Resolves up to two initials for the avatar.
                 * @param {Record<string, any>} user The user profile object.
                 * @returns {string} One or two uppercase initials, or "?".
                 */
                resolveInitials(user) {
                    const source = this.resolveDisplayName(user) || (typeof user.email === 'string' ? user.email : '')
                    const parts = source.split(/[\s@._-]+/).filter(Boolean)
                    const initials = parts.slice(0, 2).map((part) => part[0].toUpperCase()).join('')
                    return initials || '?'
                }

                /**
                 * Formats a date-like value.
                 * @param {any} value A Date, ISO string, or timestamp.
                 * @returns {string} A locale string, or the raw value on failure.
                 */
                formatDate(value) {
                    const date = value instanceof Date ? value : new Date(value)
                    return Number.isNaN(date.getTime()) ? String(value) : date.toLocaleString()
                }
            }

            /** @type {Record<string, HTMLElement>} Named DOM elements used by the widget. */
            const elements = {
                identity: document.getElementById('identity'),
                avatar: document.getElementById('avatar'),
                name: document.getElementById('name'),
                subtitle: document.getElementById('subtitle'),
                fields: document.getElementById('fields'),
                empty: document.getElementById('empty'),
            }

            /** @type {UserProfileWidget} Active widget controller. */
            const userProfileWidget = new UserProfileWidget(elements)

            void userProfileWidget.start()
        </script>
    </body>
