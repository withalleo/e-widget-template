  WIDGET SUMMARY
  ==============

  An Alleo E-widget that adds new objects (notepad, sticky note, HTML embed)
  to the board.

  Outgoing actions (triggered via alleo.triggerAction):
    - onloaded ()   fired once after init

  Incoming actions (handled via alleo.onIncomingAction):
    - none

  Board content (added via alleo.addContent):
    - notepad      (textFormat: 'markdown', text: string)
    - sticky-note  (text, color, outlineColor, shape)
    - html         (html: string)

  ⚠ REQUIRED SETTINGS: "Enable Alleo board features for enclosed content"
     and "Enable adding new content to the board".
-->
<!doctype html>
<html lang="en">
<head>
    <meta charset="UTF-8"/>
    <meta content="width=device-width, initial-scale=1.0" name="viewport"/>
    <title>Alleo E-widget — Add content</title>
    <!-- ═══════════════ Configuration ═══════════════ -->
    <script>
        /** User-tunable values. Edit these to customize the widget. */
        const settings = {
            /** @type {{type: string, textFormat: string, text: string}} Notepad object template. */
            notepadTemplate: {
                type: 'notepad',
                textFormat: 'markdown',
                text: '# New note\n\nAdded from iframe.',
            },
            /** @type {{type: string, text: string, color: string, outlineColor: string, shape: string}} Sticky-note template. */
            stickyTemplate: {
                type: 'sticky-note',
                text: 'Remember to review the draft',
                color: '#FFE680',
                outlineColor: '#B38F00',
                shape: 'square',
            },
            /** @type {{type: string, html: string}} HTML-embed object template. */
            htmlTemplate: {
                type: 'html',
                html: '<!doctype html><html><body style="background-color: black; color: white;"><h2>Hello from iframe</h2></body></html>',
            },
        }
    </script>
    <style>
        body {
            --font-family: Arial, sans-serif; /* widget font */
        }
    </style>
    <!-- ═════════════ END Configuration ═════════════ -->

<body>
<EWidgetSDK/>
<h1>Alleo E-widget — Add content</h1>
<p>These commands require "Enable adding new content to the board" in settings.</p>

<button id="add-notepad">Add notepad</button>
<button id="add-sticky">Add sticky note</button>
<button id="add-html">Add HTML embed</button>

<script>
    'use strict'

    /**
     * Adds new objects (notepad, sticky note, HTML embed) to the board via
     * alleo.addContent. Each button maps to one content template.
     */
    class AddContentWidget {
        /* ═══════════════ Configuration ═══════════════ */

        /** @type {{type: string, textFormat: string, text: string}} Notepad object template. */
        notepadTemplate = settings.notepadTemplate

        /** @type {{type: string, text: string, color: string, outlineColor: string, shape: string}} Sticky-note template. */
        stickyTemplate = settings.stickyTemplate

        /** @type {{type: string, html: string}} HTML-embed object template. */
        htmlTemplate = settings.htmlTemplate

        /* ═══════════════ End Configuration ═══════════════ */

        /**
         * @param {object} alleo The Alleo SDK messenger (the global `alleo`).
         */
        constructor(alleo) {
            /** @type {object} Alleo SDK messenger. */
            this.alleo = alleo
        }

        /**
         * Wire each button to its content template, then announce readiness.
         * @fires onloaded
         */
        init() {
            document.getElementById('add-notepad').addEventListener('click', () => this.addContent(this.notepadTemplate))
            document.getElementById('add-sticky').addEventListener('click', () => this.addContent(this.stickyTemplate))
            document.getElementById('add-html').addEventListener('click', () => this.addContent(this.htmlTemplate))

            // Listeners attached — announce the widget is ready.
            this.alleo.triggerAction('onloaded')
            console.log('AddContentWidget.init: ready')
        }

        /**
         * Add a new object to the board.
         * @param {object} params Parameters passed straight to alleo.addContent.
         */
        addContent(params) {
            this.alleo.addContent(params)
            console.log('AddContentWidget.addContent:', params.type)
        }
    }

    new AddContentWidget(alleo).init()
</script>
</body>
</html>
