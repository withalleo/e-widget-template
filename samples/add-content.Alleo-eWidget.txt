<!-- WIDGETHELP:
## What it does

This widget demonstrates adding new objects to an Alleo board from inside an
E-widget. It offers three touch-friendly buttons that each create a different
kind of board object: a Markdown **notepad**, a short **sticky note**, and a
small **HTML embed**.

## How to use it

1. Import the widget and enable board features plus **Enable adding new content
   to the board** in its settings.
2. Tap **Add notepad**, **Add sticky note**, or **Add HTML embed**.
3. The new object appears on the board, and the widget fires a `content-added`
   action carrying the object type so automations can react.

You can also create any of the three objects from another board object or
automation by sending the matching incoming action (`add-notepad`,
`add-sticky-note`, `add-html`).

## Settings

Board editors can adjust the background, primary, and text colors from the
widget settings. The object templates (notepad text, sticky note text/color,
HTML markup) live in the configuration section near the top of the file.

## Notes

The widget keeps no state and is independent per instance. It requires the
**Enable adding new content to the board** capability; without it the buttons
have no effect.
-->

        <script>
            /**
             * Adds notepad, sticky-note, and HTML-embed objects to the board via
             * alleo.addContent, and mirrors each creation as an outgoing action.
             */
            class AddContentWidget {
                /* =============== Configuration =============== */

                /** @type {Record<string, object>} Content templates keyed by button kind. */
                templates = {
                    'notepad': settings.notepadTemplate,
                    'sticky-note': settings.stickyTemplate,
                    'html': settings.htmlTemplate,
                }

                /** @type {Record<string, string>} Incoming action ids mapped to a button kind. */
                incomingActionMap = {
                    'add-notepad': 'notepad',
                    'add-sticky-note': 'sticky-note',
                    'add-html': 'html',
                }

                /** @type {string} Outgoing action fired after content is added. */
                contentAddedActionId = 'content-added'

                /* =============== End Configuration =============== */

                /**
                 * Stores DOM references used by the controller.
                 * @param {NodeListOf<HTMLButtonElement>} buttons Add-content buttons.
                 * @param {HTMLElement} statusElement Live status line.
                 */
                constructor(buttons, statusElement) {
                    /** @type {NodeListOf<HTMLButtonElement>} Add-content buttons. */
                    this.buttons = buttons
                    /** @type {HTMLElement} Live status line. */
                    this.statusElement = statusElement
                }

                /**
                 * Wires interactions, registers incoming actions, and initializes the SDK.
                 */
                async start() {
                    this.bindEvents()
                    this.bindIncomingActions()
                    await alleo.initialize()
                    console.log('[AddContentWidget.start] Widget initialized')
                }

                /**
                 * Attaches touch/click handlers to every add-content button.
                 */
                bindEvents() {
                    this.buttons.forEach((button) => {
                        button.addEventListener('click', () => this.addContent(button.dataset.kind))
                    })
                }

                /**
                 * Registers incoming actions so other board objects can add content.
                 */
                bindIncomingActions() {
                    alleo.onIncomingAction(({ actionId }) => {
                        const kind = this.incomingActionMap[actionId]
                        if (kind) this.addContent(kind)
                    })
                }

                /**
                 * Adds a board object of the requested kind and reports it.
                 * @param {string} kind Template key ('notepad' | 'sticky-note' | 'html').
                 */
                addContent(kind) {
                    const template = this.templates[kind]
                    if (!template) {
                        console.warn(`[AddContentWidget.addContent] Unknown kind: ${kind}`)
                        return
                    }

                    alleo.addContent(template)
                    alleo.triggerAction(this.contentAddedActionId, { type: kind })
                    this.statusElement.textContent = `Added a ${kind.replace('-', ' ')}.`
                    console.log(`[AddContentWidget.addContent] Added ${kind}`)
                }
            }

            /** @type {NodeListOf<HTMLButtonElement>} Add-content buttons. */
            const buttons = document.querySelectorAll('.add-button')

            /** @type {HTMLElement} Live status line. */
            const statusElement = document.getElementById('status')

            /** @type {AddContentWidget} Active widget controller. */
            const addContentWidget = new AddContentWidget(buttons, statusElement)

            void addContentWidget.start()
        </script>
