<!-- WIDGETHELP:
## What it does

This widget demonstrates secure API access from an Alleo E-widget. Press **Say “Hi!”** and the widget sends a small request to OpenAI through Alleo’s protected backend proxy, then displays the returned greeting. The OpenAI API key is never stored in this HTML file and is never exposed to the iframe.

## How to use it

1. Import this file as an Alleo E-widget.
2. Open the widget’s context menu and choose **Configure External Connection `openaiApi`**.
3. Enter an OpenAI API key with access to the configured model.
4. Press **Say “Hi!”** to send the request and display the response.

The incoming `say-hi` action can also start the request from another board object or automation.

## Settings and limitations

Board editors can adjust the background, primary, and text colors from the widget settings. The demo requires Alleo Online, internet access, protected-backend support, and a valid OpenAI API key. It does not work in offline or on-premise deployments that lack the protected backend proxy. The response is kept only in the current widget instance and is not synchronized across viewers.
-->

        <script>
            /**
             * Runs a minimal OpenAI greeting demo through Alleo's protected backend proxy.
             */
            class SecureHiWidget {
                /* =============== Configuration =============== */

                /** Protected backend connection ID used by the proxy request. */
                backendKeyId = settings.backendKeyId

                /** Exact OpenAI API endpoint called by this widget. */
                endpointUrl = settings.endpointUrl

                /** OpenAI model used for the request. */
                model = settings.model

                /** Prompt sent to OpenAI. */
                prompt = settings.prompt

                /** Instruction used to constrain the greeting response. */
                instructions = settings.instructions

                /** Maximum output-token allowance for a response. */
                maxOutputTokens = settings.maxOutputTokens

                /** Text shown on the primary action button. */
                buttonLabel = settings.buttonLabel

                /** Text shown before any request has been made. */
                idleMessage = settings.idleMessage

                /** Text shown while a request is running. */
                loadingMessage = settings.loadingMessage

                /** Outgoing action ID for a user-started request. */
                requestedActionId = settings.requestedActionId

                /** Outgoing action ID for a successful response. */
                respondedActionId = settings.respondedActionId

                /** Outgoing action ID for a failed response. */
                failedActionId = settings.failedActionId

                /** Incoming action ID that starts a greeting request. */
                incomingActionId = settings.incomingActionId

                /* =============== End Configuration =============== */

                /**
                 * Creates the widget controller and stores its required DOM elements.
                 * @param {HTMLButtonElement} buttonElement Primary action button.
                 * @param {HTMLElement} responseElement Live response display.
                 */
                constructor(buttonElement, responseElement) {
                    /** @type {HTMLButtonElement} Primary control used to start a request. */
                    this.button = buttonElement

                    /** @type {HTMLElement} Live region that displays status, responses, and errors. */
                    this.response = responseElement

                    /** @type {boolean} Prevents overlapping protected backend requests. */
                    this.isRequestPending = false

                    /** @type {null | (() => void)} Unsubscribes the incoming-action listener if needed. */
                    this.unsubscribeIncomingAction = null
                }

                /**
                 * Binds all controls, registers board integrations, and initializes the Alleo SDK.
                 */
                async start() {
                    this.bindEvents()
                    this.bindIncomingActions()
                    this.renderIdle()
                    await alleo.initialize()
                    console.log('[SecureHiWidget.start] Widget initialized')
                }

                /**
                 * Connects touch and click interaction to the greeting request.
                 */
                bindEvents() {
                    this.button.addEventListener('click', () => this.handleButtonClick())
                }

                /**
                 * Registers the incoming action that can start the demo externally.
                 */
                bindIncomingActions() {
                    this.unsubscribeIncomingAction = alleo.onIncomingAction((payload) => {
                        this.handleIncomingAction(payload)
                    })
                }

                /**
                 * Handles a user pressing the primary action button.
                 */
                handleButtonClick() {
                    console.log('[SecureHiWidget.handleButtonClick] User requested a greeting')
                    alleo.triggerAction(this.requestedActionId, { prompt: this.prompt })
                    void this.requestGreeting('user')
                }

                /**
                 * Handles commands sent from Alleo board objects or automations.
                 * @param {{ actionId?: string, data?: Record<string, unknown> }} payload Incoming action payload.
                 */
                handleIncomingAction(payload) {
                    if (payload?.actionId !== this.incomingActionId) return
                    console.log('[SecureHiWidget.handleIncomingAction] Incoming say-hi action received')
                    void this.requestGreeting('incoming-action')
                }

                /**
                 * Calls OpenAI through the protected proxy and renders the result.
                 * @param {'user' | 'incoming-action'} source Origin of the request.
                 */
                async requestGreeting(source) {
                    if (this.isRequestPending) {
                        console.warn(`[SecureHiWidget.requestGreeting] Ignored overlapping request from ${source}`)
                        return
                    }

                    this.isRequestPending = true
                    this.renderLoading()

                    try {
                        const responsePayload = await alleo.fetchProtectedUrlJSON(
                            this.backendKeyId,
                            this.endpointUrl,
                            {
                                method: 'POST',
                                headers: { 'Content-Type': 'application/json' },
                                body: JSON.stringify(this.buildRequestBody()),
                            },
                        )

                        const apiError = this.extractApiError(responsePayload)
                        if (apiError) throw new Error(apiError)

                        const responseText = this.extractResponseText(responsePayload)
                        if (!responseText) throw new Error('OpenAI returned no text response.')

                        this.renderResponse(responseText)
                        alleo.triggerAction(this.respondedActionId, { response: responseText })
                        console.log('[SecureHiWidget.requestGreeting] Greeting received')
                    } catch (error) {
                        const errorMessage = this.getErrorMessage(error)
                        this.renderError(errorMessage)
                        alleo.triggerAction(this.failedActionId, { error: errorMessage })
                        console.error('[SecureHiWidget.requestGreeting] Protected request failed:', error)
                    } finally {
                        this.isRequestPending = false
                        this.button.disabled = false
                        this.button.removeAttribute('aria-busy')
                    }
                }

                /**
                 * Builds the JSON body accepted by the OpenAI Responses API.
                 * @returns {Record<string, unknown>} Request body for the protected API call.
                 */
                buildRequestBody() {
                    return {
                        model: this.model,
                        instructions: this.instructions,
                        input: this.prompt,
                        max_output_tokens: this.maxOutputTokens,
                        store: false,
                    }
                }

                /**
                 * Extracts an API error message from a successful JSON transport response.
                 * @param {unknown} payload Parsed API response.
                 * @returns {string} API error text, or an empty string when none is present.
                 */
                extractApiError(payload) {
                    if (!payload || typeof payload !== 'object') return ''
                    const errorValue = payload.error
                    if (!errorValue || typeof errorValue !== 'object') return ''
                    return typeof errorValue.message === 'string' ? errorValue.message.trim() : ''
                }

                /**
                 * Extracts assistant text from a Responses API payload without trusting HTML.
                 * @param {unknown} payload Parsed API response.
                 * @returns {string} Combined assistant output text.
                 */
                extractResponseText(payload) {
                    if (!payload || typeof payload !== 'object') return ''

                    if (typeof payload.output_text === 'string' && payload.output_text.trim()) {
                        return payload.output_text.trim()
                    }

                    if (!Array.isArray(payload.output)) return ''

                    const textParts = []
                    for (const outputItem of payload.output) {
                        if (!outputItem || !Array.isArray(outputItem.content)) continue

                        for (const contentItem of outputItem.content) {
                            if (contentItem?.type !== 'output_text') continue
                            if (typeof contentItem.text !== 'string') continue

                            const text = contentItem.text.trim()
                            if (text) textParts.push(text)
                        }
                    }

                    return textParts.join('\n').trim()
                }

                /**
                 * Converts any thrown value into a concise message suitable for the widget.
                 * @param {unknown} error Thrown request error.
                 * @returns {string} Human-readable failure message.
                 */
                getErrorMessage(error) {
                    if (error instanceof Error && error.message.trim()) return error.message.trim()
                    if (typeof error === 'string' && error.trim()) return error.trim()
                    return 'The protected OpenAI request failed.'
                }

                /**
                 * Restores the initial prompt and interactive button state.
                 */
                renderIdle() {
                    this.button.textContent = this.buttonLabel
                    this.button.disabled = false
                    this.response.dataset.state = 'idle'
                    this.response.textContent = this.idleMessage
                }

                /**
                 * Shows progress and prevents duplicate requests.
                 */
                renderLoading() {
                    this.button.disabled = true
                    this.button.setAttribute('aria-busy', 'true')
                    this.response.dataset.state = 'loading'
                    this.response.textContent = this.loadingMessage
                }

                /**
                 * Displays a successful assistant response.
                 * @param {string} responseText Plain-text assistant output.
                 */
                renderResponse(responseText) {
                    this.response.dataset.state = 'success'
                    this.response.textContent = responseText
                }

                /**
                 * Displays a protected request failure.
                 * @param {string} errorMessage Human-readable error message.
                 */
                renderError(errorMessage) {
                    this.response.dataset.state = 'error'
                    this.response.textContent = `Request failed: ${errorMessage}`
                }
            }

            /** @type {HTMLButtonElement} Primary button element required by the widget controller. */
            const sayHiButton = document.getElementById('say-hi-button')

            /** @type {HTMLElement} Response display element required by the widget controller. */
            const aiResponse = document.getElementById('ai-response')

            /** @type {SecureHiWidget} Active widget controller instance. */
            const secureHiWidget = new SecureHiWidget(sayHiButton, aiResponse)

            void secureHiWidget.start()
        </script>
    </body>
</html>
