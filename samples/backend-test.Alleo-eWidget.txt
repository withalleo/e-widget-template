<!-- WIDGETHELP:
## What it does

This widget provides three buttons for testing public, unauthenticated API backends. It can retrieve a sample todo from JSONPlaceholder, a random quote from DummyJSON, or the current Budapest temperature from Open-Meteo. The most recent result appears in a shared response panel with the backend name, request status, duration, and selected response data.

## How to use it

1. Add the widget to an Alleo board with internet access.
2. Tap **Test Todo**, **Test Quote**, or **Test Weather**.
3. Review the response status and returned fields in the result panel.
4. Connect board automations to the `backend-tested` trigger to react to completed user tests.
5. Use the incoming actions to run any of the three tests from another board object or automation.

## Settings

Board editors can adjust the background, primary, and text colors. Endpoint URLs, labels, and the Budapest weather coordinates are grouped in the configuration section near the top of the file.

## Notes

All endpoints are public and require no API key, but the widget fetches them through Alleo's protected backend proxy (`alleo.fetchProtectedUrlJSON` with `keyId: null`) so it works from the sandboxed iframe. The board must have **Allow use of protected backends** enabled and internet access; it does not work in an offline Alleo deployment. A failed request shows an inline error and does not affect the other test buttons. Widget state is independent for each instance.
-->

    <script>
        /**
         * MultiBackendTestConsole coordinates public API tests and renders compact results.
         */
        class MultiBackendTestConsole {
            /* =============== Configuration =============== */

            /** Public todo endpoint. */
            todoEndpoint = settings.todoEndpoint

            /** Public quote endpoint. */
            quoteEndpoint = settings.quoteEndpoint

            /** Public weather endpoint. */
            weatherEndpoint = settings.weatherEndpoint

            /** Default idle message. */
            idleMessage = settings.idleMessage

            /** Loading message shown while requests are active. */
            loadingMessage = settings.loadingMessage

            /** Generic request failure message. */
            errorMessage = settings.errorMessage

            /** Maximum displayed text length. */
            maximumTextLength = settings.maximumTextLength

            /* =============== End Configuration =============== */

            /**
             * Creates the console, stores DOM references, and starts initialization.
             */
            constructor() {
                /** @type {NodeListOf<HTMLButtonElement>} Backend test buttons. */
                this.buttons = document.querySelectorAll('[data-backend]')

                /** @type {HTMLElement} Backend name display. */
                this.backendNameElement = document.getElementById('backend-name')

                /** @type {HTMLElement} Request status display. */
                this.statusElement = document.getElementById('status')

                /** @type {HTMLElement} Result details container. */
                this.detailsElement = document.getElementById('details')

                /** @type {boolean} Whether a backend request is currently active. */
                this.isLoading = false

                this.bindUserEvents()
                this.bindIncomingActions()
                this.initialize()
            }

            /**
             * Initializes Alleo after all DOM and action handlers are ready.
             */
            async initialize() {
                await alleo.initialize()
                console.log('[MultiBackendTestConsole.initialize] Widget initialized')
            }

            /**
             * Attaches touch-friendly click handlers to each backend button.
             */
            bindUserEvents() {
                this.buttons.forEach((button) => {
                    button.addEventListener('click', () => this.runBackendTest(button.dataset.backend, true))
                })
            }

            /**
             * Registers incoming actions that can run each backend test remotely.
             */
            bindIncomingActions() {
                alleo.onIncomingAction(({ actionId }) => {
                    const actionMap = {
                        'test-todo-backend': 'todo',
                        'test-quote-backend': 'quote',
                        'test-weather-backend': 'weather'
                    }
                    const backendId = actionMap[actionId]
                    if (backendId) this.runBackendTest(backendId, false)
                })
            }

            /**
             * Runs a selected backend test and updates the result panel.
             * @param {string} backendId Identifier for the selected backend.
             * @param {boolean} triggeredByUser Whether the test came from a button press.
             */
            async runBackendTest(backendId, triggeredByUser) {
                if (this.isLoading) return

                const backend = this.getBackendDefinition(backendId)
                if (!backend) return

                this.setLoadingState(true, backend.label)
                const startedAt = performance.now()

                try {
                    const payload = await alleo.fetchProtectedUrlJSON(null, backend.endpoint)
                    const result = backend.normalize(payload)
                    const durationMs = Math.round(performance.now() - startedAt)

                    this.renderResult(backend.label, 'Success', durationMs, result)
                    console.log(`[MultiBackendTestConsole.runBackendTest] ${backendId} succeeded`, result)

                    if (triggeredByUser) {
                        alleo.triggerAction('backend-tested', {
                            backend: backendId,
                            success: true,
                            durationMs,
                            result: this.stringifyResult(result)
                        })
                    }
                } catch (error) {
                    const durationMs = Math.round(performance.now() - startedAt)
                    const message = error instanceof Error ? error.message : this.errorMessage

                    this.renderResult(backend.label, 'Failed', durationMs, { Error: message }, true)
                    console.error(`[MultiBackendTestConsole.runBackendTest] ${backendId} failed`, error)

                    if (triggeredByUser) {
                        alleo.triggerAction('backend-tested', {
                            backend: backendId,
                            success: false,
                            durationMs,
                            result: message
                        })
                    }
                } finally {
                    this.setLoadingState(false, backend.label)
                }
            }

            /**
             * Returns endpoint and normalization details for a backend.
             * @param {string} backendId Identifier for the selected backend.
             * @returns {{label: string, endpoint: string, normalize: function(object): Record<string, string | number | boolean>} | null}
             */
            getBackendDefinition(backendId) {
                const backends = {
                    todo: {
                        label: 'JSONPlaceholder Todo',
                        endpoint: this.todoEndpoint,
                        normalize: (payload) => ({
                            ID: Number(payload.id),
                            Title: this.truncateText(String(payload.title ?? '')),
                            Completed: Boolean(payload.completed)
                        })
                    },
                    quote: {
                        label: 'DummyJSON Quote',
                        endpoint: this.quoteEndpoint,
                        normalize: (payload) => ({
                            ID: Number(payload.id),
                            Quote: this.truncateText(String(payload.quote ?? '')),
                            Author: this.truncateText(String(payload.author ?? 'Unknown'))
                        })
                    },
                    weather: {
                        label: 'Open-Meteo Weather',
                        endpoint: this.weatherEndpoint,
                        normalize: (payload) => ({
                            Location: 'Budapest',
                            Temperature: `${payload.current?.temperature_2m ?? '—'} ${payload.current_units?.temperature_2m ?? ''}`.trim(),
                            Time: String(payload.current?.time ?? '—')
                        })
                    }
                }

                return backends[backendId] ?? null
            }

            /**
             * Renders a backend response as labeled rows.
             * @param {string} backendName Human-readable backend name.
             * @param {string} status Request status text.
             * @param {number} durationMs Request duration in milliseconds.
             * @param {Record<string, string | number | boolean>} details Values to display.
             * @param {boolean} isError Whether the response represents an error.
             */
            renderResult(backendName, status, durationMs, details, isError = false) {
                this.backendNameElement.textContent = backendName
                this.statusElement.textContent = `${status} · ${durationMs} ms`
                this.detailsElement.textContent = ''

                Object.entries(details).forEach(([label, value]) => {
                    const row = document.createElement('div')
                    row.className = 'detail-row'

                    const term = document.createElement('dt')
                    term.textContent = label

                    const description = document.createElement('dd')
                    description.textContent = String(value)
                    if (isError) description.className = 'error'

                    row.append(term, description)
                    this.detailsElement.append(row)
                })
            }

            /**
             * Updates all controls while a request is running.
             * @param {boolean} isLoading Whether a request is active.
             * @param {string} backendName Current backend name.
             */
            setLoadingState(isLoading, backendName) {
                this.isLoading = isLoading
                this.buttons.forEach((button) => {
                    button.disabled = isLoading
                })

                if (isLoading) {
                    this.backendNameElement.textContent = backendName
                    this.statusElement.textContent = this.loadingMessage
                }
            }

            /**
             * Limits text length for compact rendering.
             * @param {string} value Text to shorten.
             * @returns {string} Shortened text.
             */
            truncateText(value) {
                if (value.length <= this.maximumTextLength) return value
                return `${value.slice(0, this.maximumTextLength - 1)}…`
            }

            /**
             * Converts result fields into a short action payload string.
             * @param {Record<string, string | number | boolean>} result Result fields.
             * @returns {string} Compact JSON representation.
             */
            stringifyResult(result) {
                return this.truncateText(JSON.stringify(result))
            }
        }

        /** Main widget instance. */
        const widget = new MultiBackendTestConsole()
    </script>
</body>
